Usage & Features

oobase32 "Hello World"                           # Standard RFC 4648 encoding
oobase32 -d "JBSWY3DPEBLW64TMMQ======"           # Decode Base32 string
oobase32 -a hex "Hello World"                    # Extended Hex alphabet encoding
oobase32 -a crockford "Hello World"              # Crockford alphabet encoding
oobase32 --no-padding "test data"                # Encode omitting padding equal signs
oobase32 -w 64 document.bin                      # Wrap encoded lines at 64 characters
oobase32 --json "Hello World"                    # Output structured JSON envelope
oobase32 --mcp                                   # Launch streaming MCP stdio server
