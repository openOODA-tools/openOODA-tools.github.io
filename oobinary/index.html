Usage & Features

oobinary -b "Hello"                              # Inspect 8-bit binary strings
oobinary --swap16 "ABCD"                         # Invert 16-bit word byte pairs (BADC)
oobinary --swap32 "ABCDEFGH"                     # Invert 32-bit word byte order (DCBAHGFE)
oobinary -m 15 --op and "Data"                   # Apply 0x0F bitwise AND mask
oobinary -a 16 -p 0 payload.bin                  # Align stream to 16-byte boundary with 0x00 padding
oobinary --json "Input"                          # Output structured JSON bit telemetry
oobinary --mcp                                   # Run Model Context Protocol stdio server
