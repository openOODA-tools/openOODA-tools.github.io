Usage & Features

oocpuinfo                                                         # Inspect CPU microarchitecture & topology
oocpuinfo -t                                                      # Inspect socket, core, thread hierarchy
oocpuinfo -f                                                      # Inspect CPU instruction set flags
oocpuinfo -f --filter avx512                                      # Filter flags matching query substring
oocpuinfo -u                                                      # Audit hardware vulnerability mitigations
oocpuinfo -a                                                      # Display all diagnostic sections
oocpuinfo --demo                                                  # Run synthetic 192-core AMD EPYC demo fixture
oocpuinfo --demo --json                                           # Output structured JSON telemetry
oocpuinfo --theme matrix                                          # Styled terminal view using matrix green theme
oocpuinfo --mcp                                                   # Run Model Context Protocol stdio server
