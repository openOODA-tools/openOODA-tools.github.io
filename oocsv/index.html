Usage & Features

oocsv --headers telemetry.csv                                     # Inspect column headers and dimensions
oocsv -w "status=online" -c "hostname,cpu_pct" telemetry.csv     # Filter rows and project columns
oocsv -w "cpu_pct>80" telemetry.csv                              # Numerical comparison filter
oocsv -d ";" -c "user,role" auth.log                             # Custom delimiter support (comma, tab, semicolon, pipe)
cat data.csv | oocsv -w "active=true" -j                         # Pipeline streaming with JSON records output
oocsv --demo                                                     # Run synthetic cluster telemetry showcase
oocsv --demo --json                                              # Output demo pipeline as JSON
oocsv --mcp                                                      # Run Model Context Protocol stdio server
