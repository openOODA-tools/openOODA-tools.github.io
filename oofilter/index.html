Usage & Features

cat server.log | oofilter -e '$1 == "ERROR"'               # Filter by column value
oofilter -n -e '$1 == "INFO" && $4 == "status=200"' app.log # Multi-column predicate with line numbers
oofilter -v -e '$3 == "management"' staff.txt              # Inverted match
cat data.csv | oofilter -c -F "," -e '$2 > 100'             # Count matching records with delimiter
oofilter -j -e '$2 == "failure"' auth.log                  # Structured JSON output
oofilter -D                                                # Interactive multi-predicate showcase
oofilter --mcp                                             # Launch Model Context Protocol stdio server
