Usage & Features

oofold -w 40 document.txt                      # Fold lines to 40 columns
cat long_lines.txt | oofold -30                # Numeric width shorthand (-30)
oofold -s -w 60 prose.txt                      # Break at spaces rather than arbitrary columns
oofold -b -w 80 binary_log.txt                 # Count raw bytes rather than visual display width
oofold -j -w 50 log.txt                        # Structured JSON metrics and folded text output
oofold -D                                      # Interactive multi-case folding showcase
oofold --mcp                                   # Launch Model Context Protocol stdio server
