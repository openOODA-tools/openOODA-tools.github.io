Usage & Features

oogzip file.txt                                  # Compress to file.txt.gz
oogzip -d -c file.txt.gz                         # Decompress to standard output
oogzip -l archive.gz                             # List compressed/uncompressed sizes and ratio
oogzip -t archive.gz                             # Verify CRC-32 and ISIZE integrity
oogzip -j -l *.gz                                # Output structured JSON metrics
oogzip -D                                        # Interactive container dissection showcase
oogzip --mcp                                     # Launch Model Context Protocol stdio server
