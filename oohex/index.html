Usage & Features

oohex binary.elf                                 # Canonical hex dump with ASCII sidebar
oohex -c 8 binary.elf                            # Format with 8 columns per row
oohex -s 0x40 -n 64 binary.elf                   # Skip to offset 64, inspect 64 bytes
oohex -p data.bin                                # Plain continuous hex stream
oohex -p data.bin | oohex -r                     # Round-trip reverse hex decoding
oohex orig.bin patched.bin                       # Dual-buffer hex delta comparison
oohex -j binary.elf                              # Structured JSON output with entropy
oohex -D                                         # Interactive demonstration showcase
oohex --mcp                                      # Launch Model Context Protocol stdio server
