Native Package Managers (Tri-Distribution Parity)

# Ubuntu / Debian (APT) curl -fsSL https://openooda-tools.github.io/oomcp/install.sh | bash -s -- --deb
# Arch Linux / Manjaro (Pacman / PKGBUILD) curl -fsSL https://openooda-tools.github.io/oomcp/install.sh | bash -s -- --pkgbuild
