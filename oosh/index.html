Install

Web Universal (Zero Dependencies)

curl -fsSL https://openooda-tools.github.io/oosh/install.sh | bash

Fedora / RHEL (DNF / RPM)

curl -fsSL https://openooda-tools.github.io/oosh/install.sh | bash -s -- --dnf
# or directly:
sudo dnf install -y https://github.com/openOODA-tools/oosh/releases/download/v1.0.1/oosh-1.0.1-1.x86_64.rpm

Ubuntu / Debian (APT / DEB)

curl -fsSL https://openooda-tools.github.io/oosh/install.sh | bash -s -- --apt
# or directly:
curl -fsSL -O https://github.com/openOODA-tools/oosh/releases/download/v1.0.1/oosh_1.0.1-1_amd64.deb && sudo apt install -y ./oosh_1.0.1-1_amd64.deb

Arch Linux / Omarchy (Pacman & PKGBUILD)

curl -fsSL https://openooda-tools.github.io/oosh/install.sh | bash -s -- --pacman
# or direct package:
sudo pacman -U https://github.com/openOODA-tools/oosh/releases/download/v1.0.1/oosh-1.0.1-1-x86_64.pkg.tar.zst
# or build via PKGBUILD:
curl -fsSL https://openooda-tools.github.io/oosh/install.sh | bash -s -- --pkgbuild

Options & Uninstallation

# Simulate installation without modifying the host
curl -fsSL https://openooda-tools.github.io/oosh/install.sh | bash -s -- --dry-run

# Clean uninstallation (lockout-safe shell revert + full purge)
oosh-uninstall                           # via locally installed helper
curl -fsSL https://openooda-tools.github.io/oosh/uninstall.sh | bash  # standalone script
curl -fsSL https://openooda-tools.github.io/oosh/install.sh | bash -s -- --uninstall
# append --purge to also wipe ~/.ooshrc, ~/.oosh_history, and ~/.oosh_profile
